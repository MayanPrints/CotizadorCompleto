⚙️ Conectar Google Sheets

Pega la URL de tu Google Apps Script Web App. Todos los datos se sincronizarán automáticamente.

🔄 Sincronización bidireccional: la app lee Y escribe en tu hoja. Al abrir la app (o pulsar 🔄) descarga los datos actualizados.

👤 Usuarios: crea una hoja llamada Usuarios con columnas Usuario · Contraseña · Nombre · Rol para gestionar accesos sin tocar el código.
Pasos para conectar:
1. Google Sheets → Extensiones → Apps Script
2. Pega el script de abajo y guarda
3. Implementar → Nueva implementación → Tipo: App web
4. Ejecutar como: Yo mismo · Acceso: Cualquier usuario
5. Copia la URL generada y pégala aquí

// ── LEER datos (GET) ──────────────────────────────
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var result = {};

  // Hojas de datos normales
  ['Pedidos','Clientes','Finanzas'].forEach(function(name) {
    var sheet = ss.getSheetByName(name);
    if (!sheet) { result[name] = []; return; }
    var vals = sheet.getDataRange().getValues();
    if (vals.length <= 1) { result[name] = []; return; }
    var headers = vals[0];
    result[name] = vals.slice(1).map(function(row) {
      var obj = {};
      headers.forEach(function(h, i) { obj[String(h)] = row[i]; });
      return obj;
    });
  });

  // Hoja Eliminados — lista de IDs borrados (compartida entre dispositivos)
  var delSheet = ss.getSheetByName('Eliminados');
  if (delSheet && delSheet.getLastRow() > 1) {
    var delVals = delSheet.getDataRange().getValues();
    result['Eliminados'] = delVals.slice(1).map(function(r){ return String(r[0]); });
  } else {
    result['Eliminados'] = [];
  }

  // Hoja Papelera — items borrados completos (JSON), para sincronizar entre dispositivos
  var trashSheet = ss.getSheetByName('Papelera');
  if (trashSheet && trashSheet.getLastRow() > 1) {
    var trashVals = trashSheet.getDataRange().getValues();
    result['Papelera'] = trashVals.slice(1).map(function(r) {
      try { return JSON.parse(String(r[1])); } catch(e) { return null; }
    }).filter(Boolean);
  } else {
    result['Papelera'] = [];
  }

  // Hoja Usuarios — para login dinámico desde Sheets
  var usersSheet = ss.getSheetByName('Usuarios');
  if (usersSheet && usersSheet.getLastRow() > 1) {
    var uVals = usersSheet.getDataRange().getValues();
    var uHeaders = uVals[0]; // Usuario, Contraseña, Nombre, Rol
    result['Usuarios'] = uVals.slice(1)
      .filter(function(r){ return r[0] && r[1]; })
      .map(function(r){
        return {
          user: String(r[0]).trim().toLowerCase(),
          pass: String(r[1]).trim(),
          name: String(r[2]||r[0]).trim(),
          role: String(r[3]||'staff').trim().toLowerCase()
        };
      });
  } else {
    result['Usuarios'] = [];
  }

  return ContentService
    .createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

// ── ESCRIBIR datos (POST) ─────────────────────────
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);

  // Acción especial: registrar un ID como eliminado en la hoja Eliminados
  if (data.action === 'markDeleted') {
    var delSheet = ss.getSheetByName('Eliminados') || ss.insertSheet('Eliminados');
    if (delSheet.getLastRow() === 0) delSheet.appendRow(['ID_Eliminado','Fecha','Hoja']);
    // Evitar duplicados
    var existing = delSheet.getLastRow() > 1
      ? delSheet.getDataRange().getValues().slice(1).map(function(r){ return String(r[0]); })
      : [];
    if (existing.indexOf(String(data.id)) === -1) {
      delSheet.appendRow([String(data.id), new Date().toLocaleDateString('es-MX'), data.sheet||'']);
    }
    // Borrar también el registro de la hoja de origen
    if (data.sheet && data.sheet !== 'Eliminados') {
      var src = ss.getSheetByName(data.sheet);
      if (src && src.getLastRow() > 1) {
        var vals = src.getDataRange().getValues();
        for (var i = 1; i < vals.length; i++) {
          if (String(vals[i][0]) === String(data.id)) { src.deleteRow(i+1); break; }
        }
      }
    }
    return ContentService
      .createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Guardar item en papelera de Sheets
  if (data.action === 'saveTrashedItem') {
    var trashSheet = ss.getSheetByName('Papelera') || ss.insertSheet('Papelera');
    if (trashSheet.getLastRow() === 0) trashSheet.appendRow(['ID','Datos_JSON','Fecha']);
    // Evitar duplicados
    var existingIds = trashSheet.getLastRow() > 1
      ? trashSheet.getDataRange().getValues().slice(1).map(function(r){ return String(r[0]); })
      : [];
    if (existingIds.indexOf(String(data.id)) === -1) {
      trashSheet.appendRow([String(data.id), JSON.stringify(data.item), new Date().toLocaleDateString('es-MX')]);
    }
    return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON);
  }

  // Eliminar item de papelera de Sheets (al restaurar o vaciar)
  if (data.action === 'removeFromTrash') {
    var trashSheet = ss.getSheetByName('Papelera');
    if (trashSheet && trashSheet.getLastRow() > 1) {
      var vals = trashSheet.getDataRange().getValues();
      for (var i = 1; i < vals.length; i++) {
        if (String(vals[i][0]) === String(data.id)) { trashSheet.deleteRow(i+1); break; }
      }
    }
    // También quitar de Eliminados (si se está restaurando)
    if (data.restore) {
      var delSheet = ss.getSheetByName('Eliminados');
      if (delSheet && delSheet.getLastRow() > 1) {
        var dvals = delSheet.getDataRange().getValues();
        for (var i = dvals.length-1; i >= 1; i--) {
          if (String(dvals[i][0]) === String(data.id)) { delSheet.deleteRow(i+1); break; }
        }
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON);
  }

  // Acciones normales: append / update / delete
  var sheet = ss.getSheetByName(data.sheet) || ss.insertSheet(data.sheet);

  if (sheet.getLastRow() === 0) {
    var headers = {
      'Pedidos':  ['ID','Fecha','Cliente','Teléfono','DTF_m','Prendas','Vinil_ml','Total','Notas','Estado'],
      'Clientes': ['ID','Nombre','Teléfono','Correo','Notas','Alta'],
      'Finanzas': ['ID','Fecha','Descripción','Monto','Tipo']
    };
    if (headers[data.sheet]) sheet.appendRow(headers[data.sheet]);
  }

  if (data.action === 'append') {
    sheet.appendRow(data.row);
  } else if (data.action === 'update') {
    var vals = sheet.getDataRange().getValues();
    for (var i = 1; i < vals.length; i++) {
      if (String(vals[i][0]) === String(data.id)) {
        data.row.forEach(function(v, j) { sheet.getRange(i+1,j+1).setValue(v); });
        break;
      }
    }
  } else if (data.action === 'delete') {
    var vals = sheet.getDataRange().getValues();
    for (var i = 1; i < vals.length; i++) {
      if (String(vals[i][0]) === String(data.id)) { sheet.deleteRow(i+1); break; }
    }
  }

  return ContentService
    .createTextOutput(JSON.stringify({ok:true}))
    .setMimeType(ContentService.MimeType.JSON);
}
